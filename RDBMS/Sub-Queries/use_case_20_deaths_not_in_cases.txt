-- Use Case 20: Find Countries Present in Deaths but Not Cases
-- Objective: Identify data inconsistencies where countries appear in one table but not another using NOT IN sub-query

CREATE DATABASE IF NOT EXISTS covid_db_subqueries_notin;
USE covid_db_subqueries_notin;

-- Table: covid_deaths
CREATE TABLE covid_deaths (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country VARCHAR(100) NOT NULL,
    report_date DATE NOT NULL,
    deaths INT DEFAULT 0,
    recovery_rate DECIMAL(5, 2) DEFAULT 0.00,
    mortality_rate DECIMAL(5, 2) DEFAULT 0.00,
    UNIQUE KEY unique_country_date (country, report_date)
);

-- Table: covid_cases
CREATE TABLE covid_cases (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country VARCHAR(100) NOT NULL,
    report_date DATE NOT NULL,
    confirmed_cases INT DEFAULT 0,
    deaths INT DEFAULT 0,
    recoveries INT DEFAULT 0,
    population BIGINT DEFAULT 0,
    UNIQUE KEY unique_country_date (country, report_date)
);

-- Step 1: Find countries in covid_deaths but NOT in covid_cases using NOT IN
SELECT DISTINCT country
FROM covid_deaths
WHERE country NOT IN (SELECT DISTINCT country FROM covid_cases)
ORDER BY country;

-- Step 2: Find countries in covid_cases but NOT in covid_deaths (reverse check)
SELECT DISTINCT country
FROM covid_cases
WHERE country NOT IN (SELECT DISTINCT country FROM covid_deaths)
ORDER BY country;

-- Step 3: Full data inconsistency report using NOT IN
SELECT 'In Deaths but not Cases' AS inconsistency_type, d.country
FROM (SELECT DISTINCT country FROM covid_deaths) d
WHERE d.country NOT IN (SELECT DISTINCT country FROM covid_cases)

UNION ALL

SELECT 'In Cases but not Deaths' AS inconsistency_type, c.country
FROM (SELECT DISTINCT country FROM covid_cases) c
WHERE c.country NOT IN (SELECT DISTINCT country FROM covid_deaths)

ORDER BY inconsistency_type, country;

-- Step 4: Alternative using NOT EXISTS for better performance on large datasets
SELECT DISTINCT d.country
FROM covid_deaths d
WHERE NOT EXISTS (
    SELECT 1 FROM covid_cases c WHERE c.country = d.country
)
ORDER BY d.country;

-- Step 5: Show total deaths recorded for countries with data inconsistency
SELECT d.country, SUM(d.deaths) AS total_deaths
FROM covid_deaths d
WHERE d.country NOT IN (SELECT DISTINCT country FROM covid_cases)
GROUP BY d.country
ORDER BY total_deaths DESC;
