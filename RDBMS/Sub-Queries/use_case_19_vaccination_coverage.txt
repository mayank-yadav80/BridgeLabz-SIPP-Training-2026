-- Use Case 19: Compare Vaccination Coverage
-- Objective: Perform comprehensive coverage analysis by combining vaccinated and unvaccinated population counts using UNION

CREATE DATABASE IF NOT EXISTS covid_db_subqueries_vaccination;
USE covid_db_subqueries_vaccination;

-- Table: covid_vaccines
CREATE TABLE covid_vaccines (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country VARCHAR(100) NOT NULL,
    report_date DATE NOT NULL,
    total_vaccinated BIGINT DEFAULT 0,
    first_dose BIGINT DEFAULT 0,
    second_dose BIGINT DEFAULT 0,
    booster_dose BIGINT DEFAULT 0,
    vaccination_rate DECIMAL(5, 2) DEFAULT 0.00,
    UNIQUE KEY unique_country_date (country, report_date)
);

-- Table: covid_population
CREATE TABLE covid_population (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country VARCHAR(100) NOT NULL UNIQUE,
    total_population BIGINT DEFAULT 0
);

-- Step 1: Show vaccinated population per country
SELECT country, total_vaccinated AS population_count, 'Vaccinated' AS category
FROM covid_vaccines
WHERE report_date = '2021-07-01'
ORDER BY country;

-- Step 2: Show unvaccinated population per country using UNION
SELECT v.country,
       v.total_vaccinated AS population_count,
       'Vaccinated' AS category
FROM covid_vaccines v
WHERE v.report_date = '2021-07-01'

UNION

SELECT p.country,
       (p.total_population - v.total_vaccinated) AS population_count,
       'Unvaccinated' AS category
FROM covid_population p
JOIN covid_vaccines v ON p.country = v.country
WHERE v.report_date = '2021-07-01'
ORDER BY country, category;

-- Step 3: Comprehensive coverage analysis with vaccination and unvaccinated counts
SELECT
    v.country,
    p.total_population,
    v.total_vaccinated,
    (p.total_population - v.total_vaccinated) AS unvaccinated,
    v.vaccination_rate,
    ROUND(((p.total_population - v.total_vaccinated) / p.total_population) * 100, 2) AS unvaccinated_pct
FROM covid_vaccines v
JOIN covid_population p ON v.country = p.country
WHERE v.report_date = '2021-07-01'
ORDER BY v.vaccination_rate DESC;

-- Step 4: UNION ALL to show both doses and coverage side by side
SELECT country, first_dose AS count, 'First Dose' AS dose_type
FROM covid_vaccines
WHERE report_date = '2021-07-01'

UNION ALL

SELECT country, second_dose, 'Second Dose'
FROM covid_vaccines
WHERE report_date = '2021-07-01'

UNION ALL

SELECT country, booster_dose, 'Booster Dose'
FROM covid_vaccines
WHERE report_date = '2021-07-01'
ORDER BY country, dose_type;
