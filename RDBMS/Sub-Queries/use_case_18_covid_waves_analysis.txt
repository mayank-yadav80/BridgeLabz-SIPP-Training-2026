-- Use Case 18: Track COVID Waves - State-wise Analysis
-- Objective: Use correlated sub-query to identify if each day represents a peak in confirmed cases for a specific state

CREATE DATABASE IF NOT EXISTS covid_db_subqueries_waves;
USE covid_db_subqueries_waves;

-- Table: covid_cases_state
CREATE TABLE covid_cases_state (
    id INT AUTO_INCREMENT PRIMARY KEY,
    state VARCHAR(100) NOT NULL,
    report_date DATE NOT NULL,
    confirmed_cases INT DEFAULT 0,
    daily_new_cases INT DEFAULT 0,
    deaths INT DEFAULT 0,
    UNIQUE KEY unique_state_date (state, report_date)
);

-- Step 1: Find the maximum daily_new_cases for each state (peak value)
SELECT state, MAX(daily_new_cases) AS peak_cases
FROM covid_cases_state
GROUP BY state;

-- Step 2: Correlated sub-query to identify peak days for Mumbai
SELECT report_date, confirmed_cases, daily_new_cases,
    CASE
        WHEN daily_new_cases = (
            SELECT MAX(cs2.daily_new_cases)
            FROM covid_cases_state cs2
            WHERE cs2.state = cs1.state
        ) THEN 'PEAK WAVE'
        WHEN daily_new_cases > (
            SELECT AVG(cs2.daily_new_cases)
            FROM covid_cases_state cs2
            WHERE cs2.state = cs1.state
        ) THEN 'Above Average'
        ELSE 'Below Average'
    END AS wave_status
FROM covid_cases_state cs1
WHERE state = 'Mumbai'
ORDER BY report_date;

-- Step 3: Correlated sub-query to identify peak days across all states
SELECT state, report_date, daily_new_cases,
    CASE
        WHEN daily_new_cases = (
            SELECT MAX(cs2.daily_new_cases)
            FROM covid_cases_state cs2
            WHERE cs2.state = cs1.state
        ) THEN 'PEAK WAVE'
        ELSE 'Normal'
    END AS wave_status
FROM covid_cases_state cs1
WHERE daily_new_cases = (
    SELECT MAX(cs2.daily_new_cases)
    FROM covid_cases_state cs2
    WHERE cs2.state = cs1.state
)
ORDER BY state, report_date;

-- Step 4: Compare each day's cases against the state's running average
SELECT state, report_date, daily_new_cases,
    (SELECT AVG(cs3.daily_new_cases)
     FROM covid_cases_state cs3
     WHERE cs3.state = cs1.state
       AND cs3.report_date <= cs1.report_date
    ) AS running_avg
FROM covid_cases_state cs1
WHERE state = 'Mumbai'
ORDER BY report_date;
