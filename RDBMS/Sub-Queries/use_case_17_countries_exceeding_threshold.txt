-- Use Case 17: Identify Countries Exceeding Case Threshold
-- Objective: Find countries where maximum confirmed cases exceed 1 million using a sub-query with IN operator

CREATE DATABASE IF NOT EXISTS covid_db_subqueries_threshold;
USE covid_db_subqueries_threshold;

-- Table: covid_cases
CREATE TABLE covid_cases (
    id INT AUTO_INCREMENT PRIMARY KEY,
    country VARCHAR(100) NOT NULL,
    report_date DATE NOT NULL,
    confirmed_cases INT DEFAULT 0,
    deaths INT DEFAULT 0,
    recoveries INT DEFAULT 0,
    population BIGINT DEFAULT 0,
    UNIQUE KEY unique_country_date (country, report_date)
);

-- Step 1: Find countries where max confirmed cases exceed 1 million using sub-query
SELECT DISTINCT country
FROM covid_cases
WHERE country IN (
    SELECT country
    FROM covid_cases
    GROUP BY country
    HAVING MAX(confirmed_cases) > 1000000
)
ORDER BY country;

-- Alternative: Using EXISTS operator
SELECT DISTINCT c1.country
FROM covid_cases c1
WHERE EXISTS (
    SELECT 1
    FROM covid_cases c2
    WHERE c2.country = c1.country
    GROUP BY c2.country
    HAVING MAX(c2.confirmed_cases) > 1000000
)
ORDER BY c1.country;

-- Find countries with max confirmed cases > 1 million and their details
SELECT c.country, MAX(c.confirmed_cases) AS max_confirmed, c.population
FROM covid_cases c
WHERE c.country IN (
    SELECT country
    FROM covid_cases
    GROUP BY country
    HAVING MAX(confirmed_cases) > 1000000
)
GROUP BY c.country, c.population
ORDER BY max_confirmed DESC;
